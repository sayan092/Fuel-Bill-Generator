CREATE TABLE USER_FILE_ARCHIVE_CONFIG (
    RULE_ID            INT           NOT NULL,
    RULE_NAME          VARCHAR(100)  NOT NULL,
    FROM_PATH          VARCHAR(500)  NOT NULL,
    TO_PATH            VARCHAR(500)  NOT NULL,
    ACTION_TYPE        VARCHAR(20)   NOT NULL,
    FILE_PATTERN       VARCHAR(100)  NULL,
    FILE_AGE_DAYS      INT           NOT NULL DEFAULT 120,
    RECURSIVE          CHAR(1)       NOT NULL DEFAULT 'Y',
    DELETE_EMPTY_DIRS  CHAR(1)       NOT NULL DEFAULT 'N',
    BATCH_SIZE         INT           NULL,
    NO_OF_BATCH_RUN    INT           NOT NULL DEFAULT 1,
    PARALLEL_DEGREE    INT           NOT NULL DEFAULT 0,
    ACTIVE             CHAR(1)       NOT NULL DEFAULT 'Y',
    LAST_RUN_DT        DATETIME      NULL,
    COMMENTS           VARCHAR(500)  NULL,
    CONSTRAINT PK_FILE_ARCHIVE_CONFIG PRIMARY KEY (RULE_ID),
    CONSTRAINT CHK_UFAC_ACTIVE CHECK (ACTIVE IN ('Y','N')),
    CONSTRAINT CHK_UFAC_RECURSIVE CHECK (RECURSIVE IN ('Y','N')),
    CONSTRAINT CHK_UFAC_DELETE_EMPTY_DIRS CHECK (DELETE_EMPTY_DIRS IN ('Y','N')),
    CONSTRAINT CHK_UFAC_ACTION_TYPE CHECK (ACTION_TYPE IN ('MOVE','COMPRESS_MOVE','DELETE_EMPTY_DIRS')),
    CONSTRAINT CHK_UFAC_FILE_AGE_DAYS CHECK (FILE_AGE_DAYS >= 0),
    CONSTRAINT CHK_UFAC_BATCH_SIZE CHECK (BATCH_SIZE IS NULL OR BATCH_SIZE >= 0),
    CONSTRAINT CHK_UFAC_NO_OF_BATCH_RUN CHECK (NO_OF_BATCH_RUN >= 0),
    CONSTRAINT CHK_UFAC_PARALLEL_DEGREE CHECK (PARALLEL_DEGREE >= 0)
);
